/* number of patterns of correct/incorrect(wrong) options(auto-gen) */
/* on: number of options, bsL: list of all correct or incorrect patterns */
/* if on>=number of patterns, pick up more than one opt from each pattern */
/* if on<number of patterns, pick up one opt from randomly chosen on-patterns */
%_rp(L):=random_permutation(L);
%__mcq_make_pattern(on, bsL):=block([l:length(bsL), k, ptL:map(length,bsL), pt], if on=0 then return([]) 
else if on>=l then while apply("+",ptL)>on do (k:rand(l)+1, if ptL[k]>1 then ptL[k]:ptL[k]-1) 
else (ptL:makelist(0, l),while apply("+", ptL)<on do (k:rand(l)+1, if ptL[k]=0 then ptL[k]:1)), return(ptL));
%__mcq_nCL:%__mcq_make_pattern(%__mcq_nC, %__mcq_CLbase);
%__mcq_nWL:%__mcq_make_pattern(%__mcq_nW, %__mcq_WLbase);
%__mcq_CLL:makelist(makelist(%__mcq_CLbase[k][j], j, 1, min(%__mcq_nCL[k], length(%__mcq_CLbase[k]))), k, 1, length(%__mcq_nCL));
%__mcq_WLL:makelist(makelist(%__mcq_WLbase[k][j], j, 1, min(%__mcq_nWL[k], length(%__mcq_WLbase[k]))), k, 1, length(%__mcq_nWL));
