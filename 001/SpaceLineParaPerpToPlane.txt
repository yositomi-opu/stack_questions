tex2(m):=sconcat("\\(", tex1(m), "\\)");
tex2L(L):=block([s:"\\(", k], for k:1 thru length(L) do s:sconcat(s, if stringp(L[k]) then L[k] else tex1(L[k]), " "), s:sconcat(s, "\\)"), return(s));
matP:random_invertible_matrix(3, [-2,-1,-1,-1,0,0,1,1,1,2]);
pp:col(matP, 1);
vv:col(matP, 2);
ww:col(matP, 3);

x0:pp[1, 1]; y0:pp[2, 1]; z0:pp[3, 1];
aa:vv[1, 1]; bb:vv[2, 1]; cc:vv[3, 1];
dd:rand([-4, -3, -2, -1, 0, 1, 2, 3, 4]);
PlaneEq:sconcat("\\(", display_polynomial_varlist_tex1(expand(aa*x+bb*y+cc*z), [x, y, z]), "=", tex1(dd), "\\)");

PtA:sconcat("\\((", tex1(sequenceify(list_matrix_entries(pp))), ")\\)");
Ceq1a:tex2L([pp+2*vv, "+p", rand([-2, -1, 2])*vv]);
Ceq1b:tex2L([pp-rand([1, 2])*vv, "+p", rand([-2, -1, 2])*vv]);
Weq1a:tex2L([pp+rand([-1, 1])*vv, "+p", rand([pp, pp-vv, pp+vv])]);
Weq1b:tex2L([pp+rand([-1, 1])*vv, "+p", rand([pp, pp-vv, pp+vv])]);
Weq2a:tex2L([rand([-2, -1, 2])*pp, "+p", rand([-2, -1, 1, 2])*vv]);
Weq2b:tex2L([vv, "+p", rand([-2, -1, 1, 2])*vv]);
Weq3a:tex2L([rand([-2, -1, 2])*vv, "+p", rand([ww, pp+vv, pp-vv])]);
Weq3b:tex2L([rand([-2, -1, 2])*vv, "+p", rand([ww, pp+vv, pp-vv])]);
