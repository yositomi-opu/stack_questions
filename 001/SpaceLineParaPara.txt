tex2(m):=sconcat("\\(", tex1(m), "\\)");
tex2L(L):=block([s:"\\(", k], for k:1 thru length(L) do s:sconcat(s, if stringp(L[k]) then L[k] else tex1(L[k]), " "), s:sconcat(s, "\\)"), return(s));
matP:random_invertible_matrix(3, [-2,-1,-1,-1,0,0,1,1,1,2]);
pp:col(matP, 1);
vv:col(matP, 2);
ww:col(matP, 3);

ParaEq:tex2L([pp, "+p", vv]);
Ceq1a:tex2L([pp+2*vv, "+p", rand([-2, -1, 2])*vv]);
Ceq1b:tex2L([pp-rand([1, 2])*vv, "+p", rand([-2, -1, 2])*vv]);
Weq1a:tex2L([pp-vv, "+p", pp+vv]);
Weq1b:tex2L([pp+vv, "+p", pp-vv]);
Weq1c:tex2L([pp, "+p", ww]);
Weq2a:tex2L([pp+ww, "+p", rand([-2, -1, 2])*vv]);
Weq2b:tex2L([2*pp,  "+p", rand([-2, -1, 2])*vv]);
Weq2c:tex2L([-pp+vv, "+p", rand([-2, -1, 2])*vv]);
Weq3a:tex2L([-vv, "+p", pp+vv]);
Weq3b:tex2L([ww, "+p", rand([pp, pp-vv])]);
Weq3c:tex2L([ww, "+p", pp+vv]);
