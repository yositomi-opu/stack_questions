/* map description for options */
%_mkop(vL, wL):=block([s:"\\(", lc:lmxchar, ln:length(vL)], lmxchar:"(",
  for k:1 thru ln do s:sconcat(s, tex1(vL[k]), "\\mapsto", tex1(wL[k]), if k<ln then ", " else ""),
  s:sconcat(s, "\\)"), lmxchar:lc, return(s));

ListAL1:[%_mkop([%_dv, %_nv], [%_dv, -%_nv])];
