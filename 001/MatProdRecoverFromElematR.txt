%_rp(L):=random_permutation(L);
tex2(s):=sconcat("\\(", tex1(s), "\\)");
aL:make_random_distribution_pm([[1, 4], [2, 3], [3, 2], [4, 1], [5, 1]]);
%_mkmat(r, c):=genmatrix(lambda([i, j], rand(aL)), r, c);
%_nr:rand([3, 4, 5]);
%_nc:rand_with_prohib(3, 5, [%_nr]); 
matA:genmatrix(lambda([i, j], rand(aL)), %_nr, %_nc);
matX1:elementary_matrix_swap(%_nr, 1, 2);
matX2:elementary_matrix_swap(%_nr, 1, 3);
matX3:elementary_matrix_swap(%_nr, 2, 3);
matY1:elementary_matrix_swap(%_nc, 1, 2);
matY2:elementary_matrix_swap(%_nc, 1, 3);
matY3:elementary_matrix_swap(%_nc, 2, 3);

%_cmod(a, p):=if mod(a, p)=0 then p else mod(a, p);

flagXY:rand([1, 2]);
flagN:rand([1, 2, 3]);
matQ:[[matX1, matX2, matX3], [matY1, matY2, matY3]][flagXY][flagN];
matQ1:[[matX1, matX2, matX3], [matY1, matY2, matY3]][flagXY][%_cmod(flagN+1, 3)];
matQ2:[[matX1, matX2, matX3], [matY1, matY2, matY3]][flagXY][%_cmod(flagN+2, 3)];
matQ3:[[matX1, matX2, matX3], [matY1, matY2, matY3]][3-flagXY][1];
matQ4:[[matX1, matX2, matX3], [matY1, matY2, matY3]][3-flagXY][2];
matQ5:[[matX1, matX2, matX3], [matY1, matY2, matY3]][3-flagXY][3];
EmatQ:tex2(matQ);
if flagXY=1 then XAS:"\\(XA\\" else XAS:"\\(AX\\)";

matAQ:if flagXY=1 then XA:matQ.matA else XA:matA.matQ;


CL1:[matA];

if flagXY=1 then %_allS:{matQ.matAQ, matQ1.matAQ, matQ2.matAQ, matAQ.matQ3, matAQ.matQ4, matAQ.matQ5}
            else %_allS:{matAQ.matQ, matAQ.matQ1, matAQ.matQ2, matQ3.matAQ, matQ4.matAQ, matQ5.matAQ};
WL1:listify(setdifference(%_allS, {matA}));
