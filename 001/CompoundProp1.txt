/* The truth values of compound propositions. */
texput("NOT", "\\lnot");
texput("AND", "\\land");
texput("OR", "\\lor");
texput("THEN", "\\rightarrow");
tex2(m):=sconcat("\\(", tex1(m), "\\)");
tex2L(L):=block([s:"\\(", k], for k:1 thru length(L) do s:sconcat(s, if stringp(L[k]) then L[k] else tex1(L[k]), " "), s:sconcat(s, "\\)"), return(s));

/* 1:false, 2:true */
vP:rand([1, 2]); vQ:rand([1, 2]);
tfP:[false, true][vP];
tfQ:[false, true][vQ];
TrueS:%__mcq_lang([["en", "True"], ["ja", "真"], ["fr", "Vrai"], ["de", "Wahr"]], %_STACK_LANG);
FalseS:%__mcq_lang([["en", "False"], ["ja", "偽"], ["fr", "Faux"], ["de", "Falsch"]], %_STACK_LANG);
sP:[FalseS, TrueS][vP];
sQ:[FalseS, TrueS][vQ];

/* and : 1 true one and 3 false ones */
andList:[];
andList:append(andList, [[tfP and tfQ, tex2L([P, AND, Q])]]);
andList:append(andList, [[(not tfP) and tfQ, tex2L([NOT, P, AND, Q])]]);
andList:append(andList, [[tfP and (not tfQ), tex2L([P, AND, NOT, Q])]]);
andList:append(andList, [[(not tfP) and (not tfQ), tex2L([NOT, P, AND, NOT, Q])]]);
/* or : 3 true ones and 1 false one */
orList:[];
orList:append(orList, [[tfP or tfQ, tex2L([P, OR, Q])]]);
orList:append(orList, [[(not tfP) or tfQ, tex2L([NOT, P, OR, Q])]]);
orList:append(orList, [[tfP or (not tfQ), tex2L([NOT, P, OR, NOT, Q])]]);
orList:append(orList, [[(not tfP) or (not tfQ), tex2L([NOT, P, OR, NOT, Q])]]);
/* then: 3 true ones and 1 false one */
thList:[];
thList:append(thList, [[(not tfP) or tfQ, tex2L([P, THEN, Q])]]);
thList:append(thList, [[tfP or tfQ, tex2L([NOT, P, THEN, Q])]]);
thList:append(thList, [[(not tfP) or (not tfQ), tex2L([P, THEN, NOT, Q])]]);
thList:append(thList, [[tfP or (not tfQ), tex2L([NOT, P, THEN, NOT, Q])]]);
/* allList:append(andList, orList, thList);*/
allList:append(andList, orList);
/* allTF:maplist(first, allList);*/
TList:[];
FList:[];
for k:1 thru length(allList) do if allList[k][1] then TList:append(TList, [allList[k][2]]) else FList:append(FList, [allList[k][2]]);
