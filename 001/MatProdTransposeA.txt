TXmat:SL[MatTL[MatL[1]]];
Ymat:SL[MatL[2]];
TYmat:SL[MatTL[MatL[2]]];
Zmat:SL[MatL[3]];
TZmat:SL[MatTL[MatL[3]]];
ABC[MatL[1]]:[r1, r2];
ABC[MatL[2]]:[r2, r3];
ABC[MatL[3]]:[r1, r3];
FixA(k):=block([], if length(ABC[1])=0 then ABC[1]:TpTpse(ABC[2]) else ABC[2]:TpTpse(ABC[1]));
FixB(k):=block([], if length(ABC[3])=0 then ABC[3]:TpTpse(ABC[4]) else ABC[4]:TpTpse(ABC[3]));
FixC(k):=block([], if length(ABC[5])=0 then ABC[5]:TpTpse(ABC[6]) else ABC[6]:TpTpse(ABC[5]));
FixA(1);
FixB(1);
FixC(1);

TB:ABC[3];
TC:ABC[5];
