    if m < 0 then return("m must be nonnegative")
    elseif m = 0 then return([[]])
    else (sub:permutations_with_repetition(L, m-1), for x in L do for r in sub do ans:endcons(cons(x, r), ans),
    return(ans)));

%_is_inj(L):=is(length(unique(L))=length(L));
