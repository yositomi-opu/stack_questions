matEcl:genmatrix(lambda([i, j], if i=j then 1 else 0), %_nc, %_nc);

HintEq2:sconcat("\\(", tex1(matPinv), tex1(matP), "=", tex1(matEcl), "\\)");
HintEq:rand([HintEq1, HintEq2]);

aL:make_random_distribution_pm([[1, 10], [2, 3], [3, 1]]);
matA:genmatrix(lambda([i, j], rand(aL)), %_nr, %_nc);
matB:matA.matP;
pvL:makelist(col(matP, k), k, 1, %_nc);
qvL:makelist(col(matPinv, k), k, 1, %_nc);
bvL:makelist(col(matB, k), k, 1, %_nc);
/* map description for question text */
%_mds:"\\(";lmxchar:"(";

%_mds:sconcat(%_mds, "\\)");

/* AP=B, A=BPinv, aj=BPinv[j] */
