%_rp(L):=random_permutation(L);
tex2(s):=sconcat("\\(", tex1(s), "\\)");
tex2L(L):=block([s:"\\(", k], for k:1 thru length(L) do s:sconcat(s, if stringp(L[k]) then L[k] else tex1(L[k]), " "), s:sconcat(s, "\\)"), return(s));

cl:4;

ijL1:%_rp(rand([[1, 4], [2, 3], [1, 3], [2, 4]]));
ijL2:rand([[1, 4], [2, 3], [1, 3], [2, 4]]);
cc1:rand([-3, -2, -1, 1, 2, 3]);
cc2:rand([-2, -1, 2, 3]);

HIDARI(m):=%__lang([["en", sconcat("multiply from the left by", tex2(m))],
 ["fr", sconcat("multiplier à gauche par", tex2(m))],
 ["de", sconcat("von links mit ... multiplizieren", tex2(m))],
 ["it", sconcat("moltiplicare a sinistra per", tex2(m))],
 ["ja", sconcat(tex2(m), "を左からかける")]]);
MIGI(m):=%__lang([
 ["en", sconcat("multiply from the right by ", tex2(m))],
 ["fr", sconcat("multiplier à droite par ", tex2(m))],
 ["de", sconcat("von rechts mit ", tex2(m), " multiplizieren")],
 ["it", sconcat("moltiplicare a destra per ", tex2(m))],
 ["ja", sconcat(tex2(m), "を右からかける")]]);
%_mkop(L):=if L[2]=1 then HIDARI(L[1]) else MIGI(L[1]);

/* 1: from Left, 2: from Right */
%_opL:%_rp([
 [kihonP(rw, ijL1[1], ijL1[2], cc1), 1],
 [kihonP(rw, ijL1[2], ijL1[1], cc1), 1],
 [kihonQ(rw, ijL1[1], cc2), 1],
 [kihonQ(rw, ijL1[2], cc2), 1],
 [kihonR(rw, ijL1[1], ijL1[2]), 1],
 [kihonP(cl, ijL1[1], ijL1[2], cc1), 2],
 [kihonP(cl, ijL1[2], ijL1[1], cc1), 2],
 [kihonQ(cl, ijL1[1], cc2), 2],
 [kihonQ(cl, ijL1[2], cc2), 2],
 [kihonR(cl, ijL1[1], ijL1[2]), 2]
]);
%_calcop(L):=if L[2]=1 then L[1].matA else matA.L[1];
AllMat:map(%_calcop, %_opL);
OpS:%_mkop(%_opL[1]);
QmatA:AllMat[1];

ListAL1:[QmatA];
