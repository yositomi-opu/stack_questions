%_nr:4;
%_nc:5;
pv:rand([[1, 2, 3], [1, 3, 4], [1, 2, 4]]);
pvC:listify(setdifference(setify(makelist(k, k, %_nc)), setify(pv)));

matF:randechelon(%_nr, %_nc, pv, [-3, -2, -1, 1, 2, 3]);
pL:make_random_distribution_pm([[3, 1], [2, 3], [1, 10], [0, 3]]);
matP:random_matrix_det_list(%_nr, pL, [-1, 1]);
matA:matP.matF;
pp:matA[2, pvC[2]];
qq:matF[1, pvC[1]];

matFq:genmatrix(lambda([i, j], if [i, j]=[1, pvC[1]] then q else matF[i, j]), %_nr, %_nc);
matAp:genmatrix(lambda([i, j], if [i, j]=[2, pvC[2]] then p else matA[i, j]), %_nr, %_nc);

%_mkop(p, q):=sconcat("\\(p= ", p, ", q= ", q, "\\)");
ListAL1:[%_mkop(pp, qq)];

/* wrong multiplication */
ListBL1:[%_mkop(pp+1, qq)];
ListBL1:endcons(%_mkop(pp-1, qq), ListBL1);

ListBL1:endcons(%_mkop(pp+2, qq), ListBL1);

ListBL2:[%_mkop(pp, qq+1)];
ListBL2:endcons(%_mkop(pp, qq-1), ListBL2);
ListBL2:endcons(%_mkop(pp, qq+2), ListBL2);
ListBL2:endcons(%_mkop(pp, qq-2), ListBL2);

ListBL3:[%_mkop(pp+1, qq-1)];
ListBL3:endcons(%_mkop(pp-1, qq+1), ListBL3);
ListBL3:endcons(%_mkop(pp-2, qq+rand([1, 2])), ListBL3);
ListBL3:endcons(%_mkop(pp+2, qq-rand([1, 2])), ListBL3);
